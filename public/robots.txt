# SEO-audit 2026-09 (P2-7): een crawler volgt volgens RFC 9309 uitsluitend de
# meest specifieke user-agent-groep. Een benoemde groep met alleen "Allow: /"
# negeert daarom de Disallows onder "User-agent: *". Daarom herhalen we het
# Disallow-blok in elke benoemde crawler-groep, zodat /portal, /admin, /api enz.
# ook voor Googlebot, Bingbot en de AI-crawlers geblokkeerd blijven.

User-agent: *
Allow: /
# Portaal & auth - geen SEO-waarde
# SEO-audit 2026-09-20 (B2-2): "$"+"/" i.p.v. een kale prefix, zodat een
# toekomstige publieke route die met dezelfde letters begint (bv.
# /portfolio) niet per ongeluk meeblokkeert.

Disallow: /portal/

Disallow: /admin/

Disallow: /account/
